7 Gibbs Low pass filter

% Gibbs Phenomenon - Low
Pass Filter
clear all;
close all;
clc;
wc = 0.5*pi;
N = 95;
alpha = (N-1)/2;
eps = 0.001;
n = 0:1:N-1;
%Ideal low pass filter
hd1 = sin(wc*(n-alpha+eps));
hd2 = pi*(n-alpha+eps);
hd = (hd1)./(hd2);
%Filter coefficient (Rectangular
Window)
hn = hd;
% Frequency response
w = 0:0.01:pi;
h = freqz(hn,1,w);
plot(w/pi, abs(h));
xlabel('Normalized Frequency
(\times\pi rad/smaple)');
ylabel('Magnitude');
title('Gibs Phenomenon in Low
Pass Filter');


8 Gibbs High pass filter

% Gibbs Phenomenon - High
Pass Filter

close all;

N = 95;
alpha = (N-1)/2;
eps = 0.001;

hd1 = sin(pi*(n-alpha+eps)) -

hd2 = pi*(n-alpha+eps);

%Filter coefficient
hn = hd;
% Frequency response
w = 0:0.01:pi;
h = freqz(hn,1,w);
plot(w/pi, abs(h));
xlabel('Normalized Frequency
(\times\pi rad/sample)');
ylabel('Magnitude');
title('Gibbs Phenomenon in High
Pass Filter');
grid on;


9 Windowing Technique

clc;
wc = 0.5*pi;
N = 55;
alpha = (N-1)/2;
eps = 0.001;
n = 0:1:N-1;
hd1 = sin(wc * (n - alpha + eps));
hd2 = pi*(n - alpha + eps);
hd = (hd1)./(hd2);
wr = boxcar(N);
hn = hd.*wr';
w = 0:0.01:pi;
h = freqz(hn, 1, w);
plot(w/pi, abs(h));
hold on;
wh = hamming(N);
hn = hd.*wh';
w = 0:0.01:pi;
h = freqz(hn,1,w);
plot(w/pi, abs(h), 'r.');
hold on;
wb = blackman(N);
hn = hd.*wb';
w = 0:0.01:pi;
h = freqz(hn,1,w);
plot(w/pi, abs(h), 'g-.');
hold on;
legend('Boxcar Window','Hamming Window', 'BlackmanWindow');
xlabel('Normalized Frequency');
ylabel('Magnitude')
hold on;


10 Kaiser Window

clc;
% Given specifications
ap = 0.1;
as = 44;
ws = 30;
wp = 20;
wfs = 100;
% Calculations
B = ws - wp;
wc = 0.5*(wp + ws);
wcr = wc*2*pi/wfs;
D = (as - 7.95)/14.36;
N = ceil((wfs*D/B) + 1);
alpha = (N-1)/2;
gamma = 0.584*(as-21)^0.4 +
0.07886*(as-21);
% Ideal low-pass filter
n = 0:1:N-1;
a1 = sin(wcr*(n-alpha));
a2 = pi*(n-alpha);
hd = a1./a2;
hd(alpha+1) = 0.5;
% Kaiser window
wk = kaiser(N,gamma);
% Windowed filter coefficients
hn = hd.*wk;
% Frequency response
w = 0:1:pi;
h = freqz(hn,1,w);
% Plot magnitude response in dB
plot(w/pi,20*log10(abs(h)));
xlabel('Normalized Frequency(\times\pi rad/sample)');
ylabel('Magnitude (dB)');
title('Kaiser Window Low-Pass FIR Filter');
grid on;

12 Down sampling

% Illustration of Downsampling
clc;
N = 30; % Length of input
sequence
n = 0:1:N-1;
x = sin(2*pi*n/20) +
sin(2*pi*n/15);
M = 3; % Downsampling factor
x1 = x(1:M:N); % Keep every M-th
sample
n1 = 1:1:N/M;
subplot(2,1,1);
stem(n,x);
xlabel('n');
ylabel('x');
title('Input Sequence');
subplot(2,1,2);
stem(n1-1,x1);
xlabel('n');
ylabel('x1');
title('Downsampled Sequence');
